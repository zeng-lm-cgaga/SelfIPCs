cmake_minimum_required(VERSION 3.22)
project(selfipcs_async_consume_demo VERSION 0.1.0 LANGUAGES CXX)

option(SELFIPCS_DEMO_USE_LEGACY "Also consume the legacy C++17 core package" OFF)
option(SELFIPCS_DEMO_ENABLE_SANITIZERS "Consume an instrumented SDK" OFF)
if(NOT TARGET SelfIPCs::async)
  find_package(SelfIPCs CONFIG REQUIRED COMPONENTS async)
endif()
if(SELFIPCS_DEMO_USE_LEGACY AND NOT TARGET EdgeRuntime::edge_runtime)
  find_package(EdgeRuntime CONFIG REQUIRED)
endif()
add_executable(async_consume_demo src/main.cpp)
target_include_directories(async_consume_demo PRIVATE include)
target_compile_features(async_consume_demo PRIVATE cxx_std_20)
set_target_properties(async_consume_demo PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(async_consume_demo PRIVATE SelfIPCs::async)
if(SELFIPCS_DEMO_USE_LEGACY)
  target_compile_definitions(async_consume_demo PRIVATE SELFIPCS_DEMO_USE_LEGACY=1)
  target_link_libraries(async_consume_demo PRIVATE EdgeRuntime::edge_runtime)
endif()
target_compile_options(async_consume_demo PRIVATE
  -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(SELFIPCS_DEMO_ENABLE_SANITIZERS OR
   (TARGET self_ipcs_async AND SELFIPCS_ENABLE_SANITIZERS))
  target_compile_options(async_consume_demo PRIVATE
    -fno-omit-frame-pointer -fsanitize=address,undefined)
  target_link_options(async_consume_demo PRIVATE
    -fno-omit-frame-pointer -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME async_consume_demo COMMAND async_consume_demo)
set_tests_properties(async_consume_demo PROPERTIES TIMEOUT 10 LABELS async)
